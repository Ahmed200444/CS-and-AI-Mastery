Explanation

Big-O describes how runtime or extra memory grows as input size n grows. Analyze the structure of the algorithm rather than memorizing labels: identify loops/recursion, account for nested work, and state meaningful best/average/worst cases when they differ.

Explanation

A stack is Last-In-First-Out (LIFO) -- the most recently added item is removed first, like a stack of plates. A queue is First-In-First-Out (FIFO) -- the first item added is the first removed, like a line of people.

Key concepts

stackqueueLIFOFIFOdeque

stack.pop()   # removes 2 (LIFO)

q.append(1); q.append(2)
q.popleft()   # removes 1 (FIFO)

stack.append("A"); stack.append("B")

        elif ch in pairs:
            if not stack or stack.pop()!=pairs[ch]: return False
    return not stack
print(balanced('(a[b])'))

Explanation

Linear search checks each element one at a time -- O(n), works on any list. Binary search repeatedly halves the search space by comparing the middle element -- O(log n), but REQUIRES the data to already be sorted.

def binary_search(sorted_arr, target):
    lo, hi = 0, len(sorted_arr) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if sorted_arr[mid] == target: return mid
        elif sorted_arr[mid] < target: lo = mid + 1
        else: hi = mid - 1
    return -1

Example

def bubble_sort(nums):
    nums=nums[:]
    for end in range(len(nums)-1,0,-1):
        swapped=False
        for i in range(end):
            if nums[i]>nums[i+1]: nums[i],nums[i+1]=nums[i+1],nums[i]; swapped=True
        if not swapped: break
    return nums
print(bubble_sort([5,1,4,2]))
def selection_sort(nums):
    nums=nums[:]
    for i in range(len(nums)):
        smallest=i
        for j in range(i+1,len(nums)):
            if nums[j]<nums[smallest]: smallest=j
        nums[i],nums[smallest]=nums[smallest],nums[i]
    return nums
print(selection_sort([5,1,4,2]))
def insertion_sort(nums):
    nums=nums[:]
    for i in range(1,len(nums)):
        value=nums[i]; j=i-1
        while j>=0 and nums[j]>value:
            nums[j+1]=nums[j]; j-=1
        nums[j+1]=value
    return nums
print(insertion_sort([5,1,4,2]))

def merge_sort(nums):
    if len(nums)<=1: return nums[:]
    mid=len(nums)//2; left=merge_sort(nums[:mid]); right=merge_sort(nums[mid:])
    out=[]; i=j=0
    while i<len(left) and j<len(right):
        if left[i]<=right[j]: out.append(left[i]); i+=1
        else: out.append(right[j]); j+=1
    return out+left[i:]+right[j:]
print(merge_sort([5,1,4,2]))
def quick_sort(nums):
    if len(nums)<=1: return nums[:]
    pivot=nums[len(nums)//2]
    left=[x for x in nums if x<pivot]
    middle=[x for x in nums if x==pivot]
    right=[x for x in nums if x>pivot]
    return quick_sort(left)+middle+quick_sort(right)
print(quick_sort([5,1,4,2,4]))

13Recursion & backtracking

Explanation

Recursion solves a problem by reducing it to smaller versions of itself. A correct recursive solution needs a base case and progress toward that base case.

Key concepts

base caserecursive statebacktrackingsearch tree

Example

def subsets(nums):
    out = []
    path = []
    def dfs(i):
        if i == len(nums):
            out.append(path.copy())
            return
        dfs(i + 1)
        path.append(nums[i])
        dfs(i + 1)
        path.pop()
    dfs(0)
    return out

print(subsets([1, 2]))
Common mistake: Forgetting to undo mutable state after exploring a branch, or writing recursion that does not move toward a base case.

Explanation

Divide and conquer splits a problem into smaller independent subproblems, solves them, then combines their results. Binary search discards half the search range each step; merge sort recursively sorts two halves and merges them.

Example

def merge_sort(nums):
    if len(nums) <= 1:
        return nums
    mid = len(nums) // 2
    left = merge_sort(nums[:mid])
    right = merge_sort(nums[mid:])
    out = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            out.append(left[i]); i += 1
        else:
            out.append(right[j]); j += 1
    return out + left[i:] + right[j:]

print(merge_sort([5, 2, 8, 1]))

def min_cost_climb(cost):
    if len(cost) <= 1:
        return 0
    prev2 = 0
    prev1 = 0
    for value in cost:
        current = value + min(prev1, prev2)
        prev2, prev1 = prev1, current
    return min(prev1, prev2)

def merge_intervals(intervals):
    if not intervals:
        return []
    intervals = sorted(intervals)
    merged = [list(intervals[0])]
    for start, end in intervals[1:]:
        if start <= merged[-1][1]:
            merged[-1][1] = max(merged[-1][1], end)
        else:
            merged.append([start, end])
    return merged

Example

def next_greater(nums):
    answer = [-1] * len(nums)
    stack = []
    for i, value in enumerate(nums):
        while stack and nums[stack[-1]] < value:
            answer[stack.pop()] = value
        stack.append(i)
    return answer

print(next_greater([2,1,2,4,3]))

Explanation

Key concepts

Example

class SimpleSet:
    def __init__(self, size=8):
        self.buckets = [[] for _ in range(size)]
    def _bucket(self, value):
        return self.buckets[hash(value) % len(self.buckets)]
    def add(self, value):
        bucket = self._bucket(value)
        if value not in bucket:
            bucket.append(value)
    def contains(self, value):
        return value in self._bucket(value)

s = SimpleSet()
s.add("python")
print(s.contains("python"), s.contains("java"))
Common mistake: Ignoring collisions.
